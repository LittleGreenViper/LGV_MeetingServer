51function _get_accurate_distance ( $lat1,
52 $lon1,
53 $lat2,
54 $lon2
55 )
56{
57 if (($lat1 == $lat2) && ($lon1 == $lon2)) { // Just a quick shortcut.
58 return 0;
59 }
60
61 $a = 6378137;
62 $b = 6356752.3142;
63 $f = 1/298.257223563; // WGS-84 ellipsiod
64 $L = ($lon2-$lon1)/57.2957795131;
65 $U1 = atan((1.0-$f) * tan($lat1/57.2957795131));
66 $U2 = atan((1.0-$f) * tan($lat2/57.2957795131));
67 $sinU1 = sin($U1);
68 $cosU1 = cos($U1);
69 $sinU2 = sin($U2);
70 $cosU2 = cos($U2);
71
72 $lambda = $L;
73 $lambdaP = $L;
74 $iterLimit = 100;
75
76 do {
77 $sinLambda = sin($lambda);
78 $cosLambda = cos($lambda);
79 $sinSigma = sqrt(($cosU2*$sinLambda) * ($cosU2*$sinLambda) + ($cosU1*$sinU2-$sinU1*$cosU2*$cosLambda) * ($cosU1*$sinU2-$sinU1*$cosU2*$cosLambda));
80 if ($sinSigma==0) {
81 return 0; // co-incident points
82 }
83
84 $cosSigma = $sinU1*$sinU2 + ($cosU1*$cosU2*$cosLambda);
85 $sigma = atan2($sinSigma, $cosSigma);
86 $sinAlpha = ($cosU1 * $cosU2 * $sinLambda) / $sinSigma;
87 $cosSqAlpha = 1.0 - $sinAlpha*$sinAlpha;
88
89 if (0 == $cosSqAlpha) {
90 return 0;
91 }
92
93 $cos2SigmaM = $cosSigma - 2.0*$sinU1*$sinU2/$cosSqAlpha;
94
95 $divisor = (16.0*$cosSqAlpha*(4.0+$f*(4.0-3.0*$cosSqAlpha)));
96
97 if (0 == $divisor) {
98 return 0;
99 }
100
101 $C = $f/$divisor;
102
103 $lambdaP = $lambda;
104 $lambda = $L + (1.0-$C) * $f * $sinAlpha * ($sigma + $C*$sinSigma*($cos2SigmaM+$C*$cosSigma*(-1.0+2.0*$cos2SigmaM*$cos2SigmaM)));
105 } while (abs($lambda-$lambdaP) > 1e-12 && --$iterLimit>0);
106
107 $uSq = $cosSqAlpha * ($a*$a - $b*$b) / ($b*$b);
108 $A = 1.0 + $uSq/16384.0*(4096.0+$uSq*(-768.0+$uSq*(320.0-175.0*$uSq)));
109 $B = $uSq/1024.0 * (256.0+$uSq*(-128.0+$uSq*(74.0-47.0*$uSq)));
110 $deltaSigma = $B*$sinSigma*($cos2SigmaM+$B/4.0*($cosSigma*(-1.0+2.0*$cos2SigmaM*$cos2SigmaM)-$B/6.0*$cos2SigmaM*(-3.0+4.0*$sinSigma*$sinSigma)*(-3.0+4.0*$cos2SigmaM*$cos2SigmaM)));
111 $s = $b*$A*($sigma-$deltaSigma);
112
113 return ( abs ( round ( $s ) / 1000.0 ) );
114}

166function _clean_meeting($meeting
167 ) {
168 // Consume only reserved flag values, preserving unrelated legacy tag contents.
169 if ('bmlt:wheelchair_accessible' === ($meeting['tag0'] ?? NULL)) {
170 $meeting['wheelchair_accessible'] = true;
171 unset($meeting['tag0']);
172 }
173 $status = $meeting['tag1'] ?? NULL;
174 if (in_array($status, ['bmlt:open', 'bmlt:closed', 'bmlt:open,closed'], true)) {
175 if ('bmlt:closed' !== $status) {
176 $meeting['is_open'] = true;
177 }
178 if ('bmlt:open' !== $status) {
179 $meeting['is_closed'] = true;
180 }
181 unset($meeting['tag1']);
182 }
183
184 $keys = array_keys($meeting) ;
185 $ret = Array();
186
187 // Get rid of "housekeeping" stuff.
188 if ( isset($meeting["id"]) ) {
189 unset($meeting["id"]);
190 }
191
192 if ( isset($meeting["radius"]) ) {
193 unset($meeting["radius"]);
194 }
195
196 if ( isset($meeting["last_modified"]) ) {
197 unset($meeting["last_modified"]);
198 }
199
200 foreach ($keys as $key) {
201 if ( !empty($meeting[$key])) {
202 $value = $meeting[$key];
203 if ( "formats" == $key || "virtual_information" == $key || "physical_address" == $key ) {
204 $value = unserialize($value);
205 if ( "physical_address" == $key ) {
206 if ( isset($value["latitude"]) ) {
207 unset($value["latitude"]);
208 }
209 if ( isset($value["longitude"]) ) {
210 unset($value["longitude"]);
211 }
212 }
213 }
214
215 switch ( $key ) {
216 case "name":
217 case "time_zone":
218 $value = strval($value);
219 break;
220
221 case "server_id":
222 case "meeting_id":
223 case "weekday":
224 case "duration":
225 $value = intval($value);
226 break;
227
228 case "latitude":
229 case "longitude":
230 $value = floatval($value);
231 break;
232 }
233
234 $ret[$key] = $value;
235 }
236 }
237
238 return $ret;
239}

256function _location_predicate( $geo_center_lng,
257 $geo_center_lat,
258 $geo_radius,
259 $predicate = "",
260 $count_only = false,
261 $select_list = "z.*",
262 $order_by = "",
263 $limit = null,
264 $offset = null
265 ) {
266 if ($predicate) {
267 $predicate = "($predicate) AND";
268 }
269
270 global $config_file_path;
271 include($config_file_path);
272
273 $center_lat = floatval($geo_center_lat);
274 $center_lng = floatval($geo_center_lng);
275 $radius_km = abs(floatval($geo_radius));
276
277 // Approximate distance in KM (kept from your original).
278 $distance_expr = "p.distance_unit
279 * DEGREES(ACOS(COS(RADIANS(p.latpoint))
280 * COS(RADIANS(z.latitude))
281 * COS(RADIANS(p.longpoint - z.longitude))
282 + SIN(RADIANS(p.latpoint))
283 * SIN(RADIANS(z.latitude))))";
284
285 // Keep the subquery so we can filter on the "distance" alias.
286 $inner = "
287 SELECT
288 $select_list,
289 p.radius,
290 p.distance_unit,
291 ($distance_expr) AS distance
292 FROM `".$_dbTableName."` AS z
293 JOIN (
294 SELECT
295 $center_lat AS latpoint,
296 $center_lng AS longpoint,
297 $radius_km AS radius,
298 111.045 AS distance_unit
299 ) AS p ON 1=1
300 WHERE z.latitude
301 BETWEEN p.latpoint - (p.radius / p.distance_unit)
302 AND p.latpoint + (p.radius / p.distance_unit)
303 AND z.longitude
304 BETWEEN p.longpoint - (p.radius / (p.distance_unit * COS(RADIANS(p.latpoint))))
305 AND p.longpoint + (p.radius / (p.distance_unit * COS(RADIANS(p.latpoint))))
306 ";
307
308 $sql = "SELECT * FROM ( $inner ) AS d WHERE $predicate (distance <= radius)";
309
310 if ($count_only) {
311 return "SELECT COUNT(*) FROM ( $sql ) AS c";
312 }
313
314 if ($order_by) {
315 $sql .= " ORDER BY $order_by";
316 }
317 if (null !== $limit) {
318 $sql .= " LIMIT ".intval($limit);
319 if (null !== $offset) {
320 $sql .= " OFFSET ".intval($offset);
321 }
322 }
323
324 return $sql;

483 ) {

497 $pdo_instance = new LGV_MeetingServer_PDO($_dbName, $_dbLogin, $_dbPassword, $_dbType, $_dbHost, $_dbPort);

531 );

534 $temp_table_exists = false;
535 $temp_table_was_promoted = true;
536
537 // Do this only after the new table is live.
538 $pdo_instance->preparedStatement("UPDATE `$_dbMetaTableName` SET `last_update`=? WHERE 1;", [time()]);
539
540 if ( !empty($old_table_name) ) {
541 try {
542 _drop_table_if_exists($pdo_instance, $old_table_name);
543 } catch (Throwable $ignored) {
544 // The old table is not dangerous to the correctness of the now-live replacement.
545 // A later maintenance pass can remove it if this drop fails.
546 }
547 }
548
549 return $number_of_meetings;
550 }
551 }
552 } catch (Throwable $throwable) {
553 echo("Error updating the data!\n");
554 var_dump($throwable);
555 } finally {
556 if ( !$temp_table_was_promoted && $temp_table_exists && (NULL !== $pdo_instance) ) {
557 try {
558 _drop_table_if_exists($pdo_instance, $tempDBName);
559 } catch (Throwable $ignored) {
560 }
561 }
562 }
563
564 return NULL;
565}
