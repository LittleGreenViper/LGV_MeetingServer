61 protected static function _uses_naws_placeholder($meeting_object, array $meeting): bool {
62 $virtual_only = isset($meeting_object->venue_type)
63 ? __VENUE_TYPE_VIRTUAL == $meeting_object->venue_type
64 : !isset($meeting['physical_location']);
65 // TC formats also identify virtual-only meetings in older BMLT data.
66 $virtual_only = $virtual_only || !isset($meeting['physical_location']);
67 if (!$virtual_only || !is_numeric($meeting_object->latitude ?? NULL) ||
68 !is_numeric($meeting_object->longitude ?? NULL)) {
69 return false;
70 }
71 $latitude = floatval($meeting_object->latitude);
72 $longitude = floatval($meeting_object->longitude);
73 if (!is_finite($latitude) || !is_finite($longitude) || abs($latitude) > 90 || abs($longitude) > 180) {
74 return false;
75 }
76 $latitude_delta = deg2rad($latitude - 34.2359868);
77 $longitude_delta = deg2rad($longitude - (-118.5634847));
78 $a = sin($latitude_delta / 2) ** 2 + cos(deg2rad(34.2359868)) *
79 cos(deg2rad($latitude)) * sin($longitude_delta / 2) ** 2;
80 $distance_meters = 6371000 * 2 * asin(sqrt(min(1, max(0, $a))));
81 return $distance_meters <= 100;
82 }
